KONFIGURASI SPREADSHEET // ========================================================================= // Jika script dibuka via 'Ekstensi > Apps Script' di Spreadsheet, biarkan KOSONG "". // Jika script dibuat terpisah di script.google.com, PASTE ID Spreadsheet di bawah: var SPREADSHEET_ID = ""; var ONLINE_THRESHOLD_MS = 3 * 60 * 1000; var CACHE_KEY_SESSIONS = "PORTAL_SESSIONS_MAP"; var CACHE_KEY_SUMMARY = "PORTAL_CACHE_SUMMARY"; /** * JALANKAN FUNGSI INI DI EDITOR UNTUK MEMBUKA OTORISASI GOOGLE! * Pilih fungsi ini di dropdown atas lalu klik tombol 'Run'. */ function TEST_DAN_OTORISASI() { var ss = resolveSpreadsheet(); Logger.log("✅ KONEKSI BERHASIL! Nama Spreadsheet: " + ss.getName()); var sheets = ss.getSheets(); for (var i = 0; i < sheets.length; i++) { Logger.log("- Tab sheet ditemukan: " + sheets[i].getName()); } return "Koneksi Database Sukses!"; } function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { var params = (e && e.parameter) ? e.parameter : {}; var callback = params.callback || ""; var action = params.action || "init"; var sid = params.sid || ("anon_" + new Date().getTime()); var isNewSession = params.is_new === "1"; var appName = params.app_name || ""; var userAgent = params.ua || "Browser Web"; try { var ss = resolveSpreadsheet(); if (action === "init") { updateUserPresence(sid); if (isNewSession) { logActivitySafe(ss, "Kunjungan Baru", "Portal Utama", sid, userAgent); } var settings = getSettingsSafe(ss); var apps = getAppsSafe(ss); var analytics = getAnalyticsSafe(ss, settings); return sendResponse({ status: "success", settings: settings, apps: apps, analytics: analytics }, callback); } if (action === "heartbeat") { updateUserPresence(sid); var settingsHb = getSettingsSafe(ss); var analyticsHb = getAnalyticsSafe(ss, settingsHb); return sendResponse({ status: "success", analytics: analyticsHb }, callback); } if (action === "click_track") { logActivitySafe(ss, "Klik Layanan", appName || "Layanan", sid, userAgent); return sendResponse({ status: "success", message: "Click logged" }, callback); } return sendResponse({ status: "error", message: "Aksi tidak valid: " + action }, callback); } catch (err) { return sendResponse({ status: "error", message: err.toString() }, callback); } } function sendResponse(data, callback) { var json = JSON.stringify(data); if (callback && callback.trim() !== "") { return ContentService.createTextOutput(callback + "(" + json + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function resolveSpreadsheet() { if (SPREADSHEET_ID && SPREADSHEET_ID.trim() !== "") { return SpreadsheetApp.openById(SPREADSHEET_ID.trim()); } var active = SpreadsheetApp.getActiveSpreadsheet(); if (active) return active; throw new Error("Spreadsheet tidak terhubung! Isi SPREADSHEET_ID di baris 11 Code.gs."); } function findSheetFuzzy(ss, keyword) { var sheets = ss.getSheets(); var kw = keyword.toLowerCase().trim(); for (var i = 0; i < sheets.length; i++) { if (sheets[i].getName().toLowerCase().trim() === kw) return sheets[i]; } for (var j = 0; j < sheets.length; j++) { if (sheets[j].getName().toLowerCase().indexOf(kw) !== -1) return sheets[j]; } return null; } function updateUserPresence(sid) { if (!sid) return; var cache = CacheService.getScriptCache(); var raw = cache.get(CACHE_KEY_SESSIONS); var map = {}; if (raw) { try { map = JSON.parse(raw); } catch (e) { map = {}; } } var now = new Date().getTime(); map[sid] = now; var cleaned = {}; for (var k in map) { if (now - map[k] < ONLINE_THRESHOLD_MS) cleaned[k] = map[k]; } cache.put(CACHE_KEY_SESSIONS, JSON.stringify(cleaned), 300); } function countOnlineUsers() { var cache = CacheService.getScriptCache(); var raw = cache.get(CACHE_KEY_SESSIONS); if (!raw) return 1; try { var map = JSON.parse(raw); var now = new Date().getTime(); var count = 0; for (var k in map) { if (now - map[k] < ONLINE_THRESHOLD_MS) count++; } return Math.max(1, count); } catch (e) { return 1; } } function getSettingsSafe(ss) { var sheet = findSheetFuzzy(ss, "Pengaturan"); var settings = {}; if (!sheet) return settings; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var key = data[i][0] ? data[i][0].toString().trim() : ""; var val = data[i][1] !== undefined ? data[i][1].toString().trim() : ""; if (key) settings[key] = val; } return settings; } function getAppsSafe(ss) { var sheet = findSheetFuzzy(ss, "Apps"); var apps = []; if (!sheet) return apps; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var title = data[i][0] ? data[i][0].toString().trim() : ""; var category = data[i][1] ? data[i][1].toString().trim() : "Layanan"; var url = data[i][2] ? data[i][2].toString().trim() : "#"; var icon = data[i][3] ? data[i][3].toString().trim() : "external-link"; var desc = data[i][4] ? data[i][4].toString().trim() : ""; var badge = data[i][5] ? data[i][5].toString().trim() : ""; var active = data[i][6] !== undefined ? data[i][6].toString().trim().toUpperCase() : "TRUE"; if (title && (active === "TRUE" || active === "1")) { apps.push({ title: title, category: category, url: url, icon: icon, desc: desc, badge: badge }); } } return apps; } function getAnalyticsSafe(ss, settings) { var cache = CacheService.getScriptCache(); var online = countOnlineUsers(); var cachedTotal = cache.get(CACHE_KEY_SUMMARY); if (cachedTotal !== null) { return { total: parseInt(cachedTotal, 10), online: online }; } var sheet = findSheetFuzzy(ss, "Analitik"); var counterAwal = settings.counter_awal ? parseInt(settings.counter_awal, 10) : 150; var total = counterAwal; if (!sheet) return { total: total, online: online }; var h2Val = sheet.getRange("H2").getValue(); if (h2Val !== "" && !isNaN(h2Val) && Number(h2Val) > 0) { total = Number(h2Val); } else { var lastRowA = getNextEmptyRowInColumn(sheet, 1) - 1; if (lastRowA >= 2) { var actRange = sheet.getRange(2, 2, lastRowA - 1, 1).getValues(); var c = 0; for (var i = 0; i < actRange.length; i++) { if (actRange[i][0] === "Kunjungan Baru") c++; } total = counterAwal + c; } } cache.put(CACHE_KEY_SUMMARY, total.toString(), 60); return { total: total, online: online }; } function logActivitySafe(ss, aktivitas, target, sid, userAgent) { var sheet = findSheetFuzzy(ss, "Analitik"); if (!sheet) return; var lock = LockService.getScriptLock(); if (!lock.tryLock(5000)) return; try { var settings = getSettingsSafe(ss); if (settings.analytics_logging && settings.analytics_logging.toUpperCase() === "FALSE") { return; } var nextRow = getNextEmptyRowInColumn(sheet, 1); sheet.getRange(nextRow, 1, 1, 5).setValues([[ new Date(), aktivitas, target, sid, userAgent || "Unknown" ]]); if (aktivitas === "Kunjungan Baru") { CacheService.getScriptCache().remove(CACHE_KEY_SUMMARY); } } catch (e) { console.error("Gagal log:", e); } finally { lock.releaseLock(); } } function getNextEmptyRowInColumn(sheet, colIndex) { var maxRows = sheet.getMaxRows(); var values = sheet.getRange(1, colIndex, maxRows, 1).getValues(); for (var i = values.length - 1; i >= 0; i--) { if (values[i][0] !== "" && values[i][0] !== null && values[i][0] !== undefined) { return i + 2; } } return 2; }
